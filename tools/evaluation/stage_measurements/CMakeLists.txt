cmake_minimum_required(VERSION 3.23)
project(ZenAssetStageMeasurements LANGUAGES CXX)
enable_testing()
get_filename_component(PROCESSOR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(stage_recorder_tests RecorderTests.cpp)
target_compile_definitions(stage_recorder_tests PRIVATE ZENASSET_STAGE_MEASUREMENTS=1)
add_executable(stage_recorder_disabled_tests DisabledTests.cpp)
foreach(target stage_recorder_tests stage_recorder_disabled_tests)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${PROCESSOR_ROOT}/src")
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    endif()
endforeach()
add_test(NAME StageRecorder COMMAND stage_recorder_tests)
add_test(NAME StageRecorderDisabled COMMAND stage_recorder_disabled_tests)

# Only this evaluation project enables private production instrumentation.
# The normal project has no profiler option, target, public header or linkage.
option(ZENASSET_STAGE_BUILD_PIPELINE "Build actual pipeline measurement probe" OFF)
option(ZENASSET_STAGE_INSTRUMENT_PIPELINE "Enable private pipeline scopes" ON)
set(ZENASSET_STAGE_DEPENDENCY_CACHE "" CACHE PATH "Read-only directory containing existing dependency *-src folders")
if(ZENASSET_STAGE_BUILD_PIPELINE)
    if(ZENASSET_STAGE_DEPENDENCY_CACHE)
        file(GLOB cached_sources LIST_DIRECTORIES true "${ZENASSET_STAGE_DEPENDENCY_CACHE}/*-src")
        foreach(source IN LISTS cached_sources)
            get_filename_component(name "${source}" NAME)
            string(REGEX REPLACE "-src$" "" name "${name}")
            string(TOUPPER "${name}" name)
            set(FETCHCONTENT_SOURCE_DIR_${name} "${source}" CACHE PATH "Existing dependency source" FORCE)
        endforeach()
        set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "Use explicit existing dependency sources" FORCE)
    endif()
    set(ZENASSET_BUILD_PROCESSOR ON CACHE BOOL "Evaluation needs the actual processor" FORCE)
    set(ZENASSET_BUILD_TESTING OFF CACHE BOOL "Keep full-suite registration in the ordinary project" FORCE)
    add_subdirectory("${PROCESSOR_ROOT}" processor)
    if(ZENASSET_STAGE_INSTRUMENT_PIPELINE)
        foreach(target ZenAssetFormat ZenAssetReader ZenAssetProcessor)
            target_compile_definitions(${target} PRIVATE ZENASSET_STAGE_MEASUREMENTS=1)
        endforeach()
    endif()
    add_executable(stage_measurements main.cpp)
    target_compile_features(stage_measurements PRIVATE cxx_std_20)
    target_compile_definitions(stage_measurements PRIVATE ZENASSET_STAGE_MEASUREMENTS=1
        ZENASSET_STAGE_PIPELINE_INSTRUMENTED=$<BOOL:${ZENASSET_STAGE_INSTRUMENT_PIPELINE}>)
    target_include_directories(stage_measurements PRIVATE "${PROCESSOR_ROOT}/src" "${PROCESSOR_ROOT}/tests")
    target_link_libraries(stage_measurements PRIVATE ZenAsset::Processor ZenAsset::Reader)
    if(MSVC)
        target_compile_options(stage_measurements PRIVATE /W4 /permissive- /utf-8)
    endif()
endif()
